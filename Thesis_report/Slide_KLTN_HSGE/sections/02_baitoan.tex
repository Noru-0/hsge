\subsection{Phát biểu bài toán và phạm vi}
\begin{frame}{Phát biểu bài toán và phạm vi}
  % 2x2 grid laid out as two row-wise `columns` (not two stacked columns): this is
  % what keeps the second row aligned on both sides when the two blocks of the first
  % row wrap to a different number of lines.
  \begin{columns}[t]
    \begin{column}{0.485\textwidth}
      \uncover<1->{
        \begin{block}{Bài toán}
          \begin{itemize}\ilist
            \item \textbf{Đầu vào}: 1 website mục tiêu, \kw{0 nhãn người}.
            \item \textbf{Đầu ra}: dữ liệu trajectory $+$ \kw{SLM ${<}10$B} chuyên biệt.
          \end{itemize}
        \end{block}
      }

    \end{column}

    \hfill
    \begin{column}{0.485\textwidth}
      \uncover<2->{
        \begin{block}{Tiêu chí đánh giá}
          \begin{itemize}\ilist
            \item \textbf{WebArena}: 655 tác vụ / 4 website thực.
            \item \kw{Task SR} (hàm chấm benchmark); phụ: \kw{chi phí API / trajectory}.
          \end{itemize}
        \end{block}
      }
    \end{column}
  \end{columns}

  \gap
  \begin{columns}[t]
    \begin{column}{0.485\textwidth}
      \uncover<3->{
        \begin{block}{Ràng buộc}
          \begin{itemize}\ilist
            \item Ngân sách API mức \textbf{nhóm nhỏ}.
            \item Chạy \textbf{on-premise}: dữ liệu không rời tổ chức.
          \end{itemize}
        \end{block}
      }
    \end{column}

    \hfill
    \begin{column}{0.485\textwidth}
      \uncover<4->{
        \begin{block}{Phạm vi \& ứng dụng đích}
          \begin{itemize}\ilist
            \item 1 website $\rightarrow$ 1 agent chuyên biệt, tại chỗ.
            \item Tổng quát hóa ở mức \kw{pipeline}, không phải \kw{mô hình}.
          \end{itemize}
        \end{block}
      }
    \end{column}
  \end{columns}
\end{frame}

\subsection{Bốn thách thức cốt lõi}
\begin{frame}{Bốn thách thức cốt lõi của bài toán}
  \begin{columns}[t]
    \begin{column}{0.485\textwidth}
      \uncover<1->{
        \begin{block}{T1. Chi phí gán nhãn quá cao}
          \begin{itemize}\ilist
            \item Gán nhãn tay: \textbf{AutoWebGLM} \kw{20 người $\times$ 1 tháng}.
            \item Sinh ngoại tuyến rẻ (\kw{${\sim}3\%$}) nhưng lệch môi trường: \kw{6\%} vs \kw{16\%}.
          \end{itemize}
        \end{block}
      }
    \end{column}

    \hfill
    \begin{column}{0.485\textwidth}
      \uncover<2->{
        \begin{block}{T2. Tri thức thu được bị phân mảnh}
          \begin{itemize}\ilist
            \item Trajectory thô rời rạc, \kw{không có đơn vị kỹ năng}.
            \item Thao tác lặp ở nhiều trang vẫn là dữ liệu mới.
          \end{itemize}
        \end{block}
      }
    \end{column}
  \end{columns}

  \gap
  \begin{columns}[t]
    \begin{column}{0.485\textwidth}
      \uncover<3->{
        \begin{block}{T3. Dương tính giả khi agent tự chấm}
          \begin{itemize}\ilist
            \item Không nhãn người $\rightarrow$ agent \kw{tự khai hoàn thành}.
            \item Dữ liệu bẩn \kw{dạy mô hình khai gian}.
          \end{itemize}
        \end{block}
      }
    \end{column}

    \hfill
    \begin{column}{0.485\textwidth}
      \uncover<4->{
        \begin{block}{T4. Khó ghép thành tác vụ dài}
          \begin{itemize}\ilist
            \item Tác vụ tự sinh \kw{bó hẹp trong một trang}.
            \item Thiếu cơ chế \textit{A xong mới làm được B}.
          \end{itemize}
        \end{block}
      }
    \end{column}
  \end{columns}
\end{frame}
