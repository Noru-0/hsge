\subsection{Phát biểu bài toán và phạm vi}
\begin{frame}{Phát biểu bài toán và phạm vi}
  \begin{columns}[t]
    \begin{column}{0.485\textwidth}
      \uncover<1->{
        \begin{block}{Bài toán}
          \begin{itemize}\ilist
            \item \textbf{Đầu vào}: 1 website mục tiêu, \kw{0 nhãn con người}.
            \item \textbf{Đầu ra}: dữ liệu trajectory $+$ \kw{SLM ${<}10$B} chuyên biệt.
          \end{itemize}
        \end{block}
      }
    \end{column}

    \hfill
    \begin{column}{0.485\textwidth}
      \uncover<2->{
        \begin{block}{Phạm vi \& ứng dụng đích}
          \begin{itemize}\ilist
            \item 1 website $\rightarrow$ 1 agent chuyên biệt.
            \item Ngân sách API ở mức \kw{nhóm nghiên cứu nhỏ}.
            \item Tổng quát hóa \kw{pipeline}, không phải mô hình.
          \end{itemize}
        \end{block}
      }
    \end{column}
  \end{columns}
\end{frame}

\subsection{Bốn thách thức cốt lõi}
\begin{frame}{Bốn thách thức cốt lõi của bài toán}
  \begin{columns}[t]
    \begin{column}{0.485\textwidth}
      \uncover<1->{
        \begin{block}{1. Thu thập dữ liệu chất lượng cao}
          \begin{itemize}\ilist
            \item Dữ liệu tự sinh dễ \kw{lặp lại, không hợp lệ}.
            \item Agent dễ tập trung hành vi quen thuộc.
            \item $\rightarrow$ Cần tự động sinh mẫu vừa \kw{hợp lệ, vừa đa dạng}.
          \end{itemize}
        \end{block}
      }
    \end{column}

    \hfill
    \begin{column}{0.485\textwidth}
      \uncover<2->{
        \begin{block}{2. Thiếu hiểu biết trước về môi trường}
          \begin{itemize}\ilist
            \item Tri thức gián tiếp \kw{không bám web mục tiêu} ($6\%$ Synatra vs $16\%$ NNetNav).
            \item $\rightarrow$ Cần \kw{tự động khám phá} và tích lũy tri thức trực tiếp trên môi trường thực.
          \end{itemize}
        \end{block}
      }
    \end{column}
  \end{columns}

  \gap
  \begin{columns}[t]
    \begin{column}{0.485\textwidth}
      \uncover<3->{
        \begin{block}{3. Sàng lọc trajectory tự sinh}
          \begin{itemize}\ilist
            \item Agent vừa sinh vừa chạy $\rightarrow$ dễ \kw{tự khai xong giả} (dương tính giả).
            \item $\rightarrow$ Cần sàng lọc bằng \kw{tín hiệu khách quan} từ môi trường trước khi train.
          \end{itemize}
        \end{block}
      }
    \end{column}

    \hfill
    \begin{column}{0.485\textwidth}
      \uncover<4->{
        \begin{block}{4. Tạo tác vụ có ngữ nghĩa}
          \begin{itemize}\ilist
            \item Task tự sinh cần mạch lạc, đúng luồng nghiệp vụ website (nhiều bước).
            \item $\rightarrow$ Cần \kw{tự động phối hợp kỹ năng} đã khám phá thành tác vụ có ý nghĩa.
          \end{itemize}
        \end{block}
      }
    \end{column}
  \end{columns}
\end{frame}
