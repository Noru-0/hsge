\subsection{Phát biểu bài toán và phạm vi}
\begin{frame}{Phát biểu bài toán và phạm vi}
  \begin{columns}[onlytextwidth,t]
    \begin{column}{0.485\textwidth}
      \uncover<1->{
        \begin{hblock}{Bài toán đặt ra}{\bhB}
          \begin{itemize}\ilist
            \item \textbf{Đầu vào}: $1$ website mục tiêu $+$ $1$ LLM API, \kw{0 nhãn người}.
            \item \textbf{Mục tiêu}: dùng LLM \kw{sinh dữ liệu} $\rightarrow$ fine-tune SLM.
            \item \textbf{Đầu ra}: tập SFT $+$ \kw{SLM ${<}10$B} chuyên biệt.
          \end{itemize}
        \end{hblock}
      }
    \end{column}

    \hfill
    \begin{column}{0.485\textwidth}
      \uncover<2->{
        \begin{hblock}{Phạm vi \& ứng dụng đích}{\bhB}
          \begin{itemize}\ilist
            \item $1$ website $\rightarrow$ $1$ agent chuyên biệt, tại chỗ.
            \item Ngân sách API mức \kw{nhóm nhỏ}.
            \item Tổng quát hóa \kw{pipeline}, không phải mô hình.
            \item Chỉ \kw{SFT/LoRA}, không dùng RL.
          \end{itemize}
        \end{hblock}
      }
    \end{column}
  \end{columns}

  \vspace{0.15cm}
  \uncover<3->{
    \begin{alertblock}{Định hướng giải pháp}
      \centering \vspace{0.03cm}
      {\footnotesize Dùng \kw{LLM mạnh} tự khám phá chính website mục tiêu, sinh trajectory để fine-tune \kw{SLM cục bộ}.}
      \vspace{0.03cm}
    \end{alertblock}
  }
\end{frame}

\subsection{Nguồn dữ liệu huấn luyện}
\begin{frame}{Dữ liệu huấn luyện SLM lấy từ đâu?}
  \begin{columns}[onlytextwidth,t]
    \begin{column}{0.485\textwidth}
      \uncover<1->{
        \begin{hblock}{Chưng cất tri thức: LLM làm thầy}{\bhG}
          \begin{itemize}\ilist
            \item LLM mạnh (API) tự thao tác trên \kw{website mục tiêu}.
            \item Mỗi task giải xong $\rightarrow$ $1$ \kw{trajectory}: lời giải mẫu của thầy.
            \item SLM \kw{học bắt chước} trajectory đó qua fine-tune.
          \end{itemize}
        \end{hblock}
      }
    \end{column}

    \hfill
    \begin{column}{0.485\textwidth}
      \uncover<2->{
        \begin{hblock}{Vì sao chưng cất là hợp lý}{\bhG}
          \begin{itemize}\ilist
            \item Trả phí API \kw{một lần}; khi chạy chỉ còn SLM cục bộ.
            \item Thầy tự ra \kw{cả đề lẫn lời giải} $\rightarrow$ không cần nhãn người.
            \item Tri thức gắn với \kw{đúng giao diện} của website mục tiêu.
          \end{itemize}
        \end{hblock}
      }
    \end{column}
  \end{columns}

  \gap
  \uncover<3->{
    \begin{alertblock}{Bài toán học của SLM}
      \centering \vspace{0.03cm}
      {\footnotesize LLM \kw{làm mẫu} cách thao tác $\;\Longrightarrow\;$ SLM học \kw{chọn hành động kế tiếp} từ trạng thái trang.}
      \vspace{0.03cm}
    \end{alertblock}
  }
\end{frame}
