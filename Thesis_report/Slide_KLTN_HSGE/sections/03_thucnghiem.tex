\section{Thực nghiệm và kết quả}
%%% ============================================================

\begin{frame}{Thiết lập thực nghiệm và câu hỏi nghiên cứu}
  \begin{columns}[t]
    \begin{column}{0.52\textwidth}
      \footnotesize
      \textbf{Môi trường.} Bốn website WebArena tự triển khai --- Shopping,
      Shopping Admin, GitLab, Reddit. Đánh giá trên \kw{655 task chính thức}
      của benchmark, chấm bằng bộ chấm tự động, ngân sách 15 bước/task,
      \texttt{temperature} $=0$.

      \vspace{0.2cm}
      \textbf{Khám phá.} Bốn đợt, mỗi đợt $\approx$21 giờ trên một website.
      Mỗi vai trò LLM dùng một mô hình riêng (solver, page\_explorer,
      episode\_judge, segmenter, operator\_inducer\ldots), tổng
      \kw{20\,808 lời gọi}.

      \vspace{0.2cm}
      \textbf{Huấn luyện.} Qwen3.5-9B $+$ LoRA (rank 16, $\alpha$ 32, mọi lớp
      tuyến tính), LLaMA-Factory, 3 epoch $=$ 804 bước, ngữ cảnh 4096 token;
      chọn \texttt{checkpoint-536} (3 giờ 05 phút trên H100-80GB).
    \end{column}
    \begin{column}{0.46\textwidth}
      \footnotesize
      \begin{block}{Bốn câu hỏi nghiên cứu}
        \textbf{CH1.} Pipeline thu được gì trên một website chưa có nhãn, và
        bao nhiêu sống sót qua các cổng chất lượng?

        \textbf{CH2.} Dữ liệu đó có nâng được năng lực của SLM không, và phần
        nâng lên \emph{thuộc về năng lực nào}?

        \textbf{CH3.} Kết quả đứng ở đâu so với các hệ thống cùng phân khúc
        dưới 10B?

        \textbf{CH4.} Thành phần nào đóng góp bao nhiêu, điểm nghẽn còn lại
        ở đâu?
      \end{block}
      \textit{Nói rõ ngay:} CH4 chỉ được trả lời \emph{một phần} --- các
      ablation có kiểm soát đã thiết kế nhưng chưa chạy; phần trả lời đến từ
      phân tích lỗi và đối chiếu liên-website.
    \end{column}
  \end{columns}
\end{frame}

        \item Task GOAL khó hơn hẳn task cục bộ: 37,9\% so với 42,0\% và 52,0\%.
        \item Cổng chấm lại loại \textbf{497} lần agent tự đánh giá sai; cổng
              hành động hữu hiệu loại thêm \textbf{104} trajectory.
      \end{itemize}

\begin{frame}{CH2--CH3 --- Phần tăng thuộc về năng lực nào, và đứng ở đâu}
  \begin{columns}[t]
    \begin{column}{0.50\textwidth}
      \footnotesize
      \begin{alertblock}{Giới hạn diễn giải quan trọng nhất}
        Mô hình nền hỏng \emph{định dạng đầu ra} ở \kw{170/655 (26,0\%)}
        episode; mô hình sau fine-tune chỉ 2/655.
        \begin{itemize}
          \item Trên đúng 170 task đó: nền \textbf{0}, fine-tune \textbf{35}.
          \item Trên 485 task nền chạy sạch: 87 vs 101,
                \textbf{$p = 0{,}120$ --- chưa đạt ý nghĩa}.
        \end{itemize}
        $\Rightarrow$ \textbf{35 trong 49} task tăng thêm thuộc về việc học
        \emph{tuân thủ giao diện hành động}; phần còn lại tuy cùng chiều nhưng
        chưa tách khỏi nhiễu.
      \end{alertblock}
      {\scriptsize Kèm một điều chỉnh thủ công phía mô hình nền (loại thành công
