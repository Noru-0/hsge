\begin{frame}{Thiết lập thực nghiệm và câu hỏi nghiên cứu}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \footnotesize
      \textbf{Thiết lập thực nghiệm}
      \begin{itemize}\setlength{\itemsep}{3pt}
        \item \textbf{Môi trường}: 4 website WebArena tự triển khai (Shopping, Shopping Admin, GitLab, Reddit).
        \item \textbf{Đánh giá}: \kw{655 task chính thức}, chấm tự động, 15 bước/task, \texttt{temperature} $=0$.
        \item \textbf{Khám phá}: 4 đợt ($\approx$21h/site), phân tách rõ vai trò các LLM agent (\kw{20\,808 lượt gọi}).
        \item \textbf{Huấn luyện}: Qwen3.5-9B $+$ LoRA, 1119 bước (5h09); chọn \texttt{checkpoint-315} (eval\_loss $=0{,}4219$).
      \end{itemize}
    \end{column}
    \begin{column}{0.50\textwidth}
      \begin{block}{Bốn câu hỏi nghiên cứu (RQ)}
        \footnotesize
        \begin{itemize}\setlength{\itemsep}{2pt}
          \item[\textbf{CH1.}] Pipeline thu được gì và chi phí bao nhiêu?
          \item[\textbf{CH2.}] SLM cải thiện năng lực nào sau fine-tune?
          \item[\textbf{CH3.}] Vị thế so với các hệ thống $<10$B hiện nay?
          \item[\textbf{CH4.}] Đóng góp từng thành phần \& điểm nghẽn?
        \end{itemize}
      \end{block}

      \vspace{0.05cm}
      {\scriptsize\textit{*CH4 được giải đáp qua phân tích lỗi và đối chiếu liên-website (ablation đầy đủ là hướng phát triển).}}
    \end{column}
  \end{columns}
\end{frame}

\subsection{CH3: Phân tích năng lực \& So sánh hệ thống}
\begin{frame}{CH2--CH3 --- Phân tích năng lực và so sánh hệ thống}
  \begin{columns}[t]
    \begin{column}{0.50\textwidth}
      \footnotesize
      \begin{alertblock}{Năng lực cốt lõi đạt được}
        \scriptsize
        Mô hình nền bị lỗi \textbf{định dạng đầu ra} ở \kw{170/655 (26,0\%)} episode; sau fine-tune chỉ còn \kw{4/655 (0,6\%)}.
        \begin{itemize}\setlength{\itemsep}{1.5pt}
          \item Nâng cao khả năng kiên trì giải task (trung vị số bước thất bại tăng từ 8 lên 13).
          \item Tăng từ 87 lên \textbf{161 task giải thành công} ($+74$ task).
        \end{itemize}
        $\Rightarrow$ SLM làm chủ hoàn toàn \kw{giao diện hành động} và cấu trúc tương tác POMDP của web.
      \end{alertblock}
    \end{column}
    \begin{column}{0.48\textwidth}
      \scriptsize
      \begin{table}
        \centering
        \begin{tabular}{@{}L{3.6cm} R{2.2cm}@{}}
          \toprule
          \textbf{Mô hình} & \textbf{Tổng 4 site (\%)} \\
          \midrule
          GPT-4o-mini & 20,0 \\
          \midrule
          Qwen-2.5-7B-Instruct & 8,3 \\
          NNetNav-7B & 19,1 \\

        \item Dẫn đầu nhóm mô hình mở dưới 10B trên WebArena 4 website.
