\subsection{Bối cảnh \& Phát biểu bài toán}
\begin{frame}{Bối cảnh, bài toán và thách thức}
  \framesubtitle{Nút thắt nằm ở mô hình, hay ở dữ liệu?}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \textbf{Bối cảnh}
      \begin{itemize}[<+->]\ilist
        \item \textbf{Web agent}: tự động hóa trình duyệt.
        \item \textbf{LLM lớn qua API}: đắt, chậm, rủi ro riêng tư.
        \item \textbf{SLM ${<}10$B}: chạy cục bộ $\rightarrow$ phải fine-tune.
        \item \kw{Nút thắt}: nhãn tay --- AutoWebGLM \textbf{20 người $\times$ 1 tháng}.
      \end{itemize}
    \end{column}
    \hfill
    \begin{column}{0.48\textwidth}
      \uncover<5->{
        \begin{block}{Câu hỏi bài toán}
          \begin{itemize}\isub
            \item Tinh chỉnh SLM cho \textbf{một website đang chạy}, \kw{0 nhãn người}?
            \item \textit{Phạm vi}: một tập website cố định --- \textbf{không} phải agent vạn năng.
          \end{itemize}
        \end{block}
      }

      \gap
      \uncover<6->{
        \textbf{4 thách thức}
        \begin{enumerate}\isub
          \item<6-> Chi phí gán nhãn.
          \item<7-> Tri thức môi trường.
          \item<8-> Báo đỗ ảo (\textit{false positive}).
          \item<9-> Tác vụ đa bước.
        \end{enumerate}
      }
    \end{column}
  \end{columns}
\end{frame}

\subsection{Mục tiêu \& Định vị nghiên cứu}
\begin{frame}{Mục tiêu và định vị so với công trình liên quan}
  \framesubtitle{Đầu ra là một agent chuyên biệt, hay một model vạn năng?}
  \begin{columns}[t]
    \begin{column}{0.44\textwidth}
      \textbf{5 mục tiêu nghiên cứu}
      \begin{itemize}[<+->]\isub
        \item[\textbf{O1}] Khám phá website chưa có nhãn.
        \item[\textbf{O2}] Sinh semantic task đã kiểm chứng.
        \item[\textbf{O3}] Tinh lọc dataset \& prompt parity.
        \item[\textbf{O4}] Fine-tune SLM \& đánh giá WebArena.
        \item[\textbf{O5}] Đúc kết best practice vận hành.
