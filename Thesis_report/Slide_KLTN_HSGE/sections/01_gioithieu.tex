\begin{frame}{Động lực của khóa luận}

      \uncover<1->{
        \begin{hblock}{Khoảng cách hiệu năng của Web Agent}{\bhA}
          \begin{itemize}\ilist
            \item \textbf{Kỳ vọng}: web agent tự hành theo lệnh ngôn ngữ tự nhiên.
            \item \textbf{WebArena}: GPT-4 chỉ đạt \kw{14,41\%} (con người đạt \kw{78,24\%}).

        \end{hblock}
      }

            \item \textbf{Rào cản của LLM}: chi phí, độ trễ, riêng tư.
            \item \textbf{SLM ${<}10$B cục bộ}: rẻ, tự chủ, bảo mật $\rightarrow$ \kw{bắt buộc fine-tune}.

          \end{itemize}

      }
