\subsection{Bối cảnh \& Phát biểu bài toán}
\begin{frame}{Bối cảnh, bài toán và thách thức}
  \begin{columns}[t]
    \begin{column}{0.49\textwidth}
      \footnotesize
      \textbf{Bối cảnh \& Động lực}
      \begin{itemize}\setlength{\itemsep}{2.5pt}
        \item \textbf{Web Agent \& LLM lớn}: Tự động hóa tác vụ web nhưng phụ thuộc API đám mây $\rightarrow$ Chi phí cao, độ trễ lớn, rủi ro bảo mật dữ liệu.
        \item \textbf{SLM (${<}10$B)}: Chi phí thấp, chạy cục bộ $\rightarrow$ Cần fine-tune (SFT) để chuyên biệt hóa cho từng website.
        \item \kw{Nút thắt dữ liệu}: Gán nhãn thủ công rất tốn kém (AutoWebGLM: \textbf{20 người} $\times$ \textbf{1 tháng}). Dữ liệu tổng hợp giúp giảm \textbf{$\approx$97\%} chi phí mỗi mẫu (Synatra).
        \item $\Rightarrow$ \textbf{Giải pháp}: Tự động hóa quá trình tạo dữ liệu từ website mục tiêu.
      \end{itemize}
    \end{column}
    \begin{column}{0.49\textwidth}
      \begin{block}{Phát biểu bài toán}
        \scriptsize
        Cho website đang hoạt động và một pretrained SLM, tự động tạo dữ liệu để fine-tune SLM hoàn thành các tác vụ web bằng ngôn ngữ tự nhiên mà \textbf{không dùng dữ liệu gán nhãn thủ công}.
      \end{block}

      \vspace{0.08cm}
      {\footnotesize\textbf{Bốn thách thức cốt lõi}}
      \begin{enumerate}\scriptsize\setlength{\itemsep}{1.5pt}
        \item \textbf{Thu thập dữ liệu}: Chi phí gán nhãn thủ công quá lớn để mở rộng.
        \item \textbf{Tri thức môi trường}: Khó chuyển giao giữa các website khác nhau.
        \item \textbf{Kiểm chứng đúng đắn}: Ngăn chặn việc tự nhận hoàn thành ảo.
        \item \textbf{Tác vụ có ngữ nghĩa}: Ghép nối thành chuỗi đa bước thực tế.
      \end{enumerate}
    \end{column}
  \end{columns}
\end{frame}
