\begin{frame}{Bối cảnh, bài toán và thách thức}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \small
      \textbf{Bối cảnh \& Động lực}
      \begin{itemize}\setlength{\itemsep}{4pt}
        \item \textbf{Web Agent \& LLM lớn}: Phụ thuộc API đám mây $\rightarrow$ Chi phí cao, độ trễ lớn, rủi ro riêng tư dữ liệu.
        \item \textbf{SLM (${<}10$B)}: Chạy cục bộ, chi phí thấp, kiểm soát trọng số $\rightarrow$ Cần fine-tune để chuyên biệt hóa.
        \item \kw{Nút thắt dữ liệu}: Gán nhãn thủ công rất tốn kém (AutoWebGLM: \textbf{20 người} $\times$ \textbf{1 tháng}).
        \item Dữ liệu tổng hợp giúp giảm \textbf{$\approx$97\%} chi phí mỗi mẫu (Synatra).
      \end{itemize}
    \end{column}
    \begin{column}{0.50\textwidth}
      \begin{block}{Phát biểu bài toán}
        \footnotesize
        Tự động khám phá website đang hoạt động và tinh chỉnh SLM hoàn thành các task ngôn ngữ tự nhiên \textbf{không dùng dữ liệu do con người gán nhãn}.
      \end{block}

      \vspace{0.1cm}
      {\footnotesize\textbf{Năm thách thức chính}}
      \begin{enumerate}\scriptsize\setlength{\itemsep}{2pt}
        \item \textbf{Khám phá thực tế}: Xử lý trang trùng cấu trúc, trang lỗi.
        \item \textbf{Sinh task ý nghĩa}: Tránh duyệt ngẫu nhiên vô nghĩa.
        \item \textbf{Tự kiểm chứng}: Đánh giá trajectory tự động, khách quan.
        \item \textbf{Ghép task dài hạn}: Mở rộng qua chuỗi nhiều trang/thao tác.
        \item \textbf{Prompt parity}: Đồng nhất giữa huấn luyện và suy luận.
      \end{enumerate}
    \end{column}
  \end{columns}
\end{frame}
