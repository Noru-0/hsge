\section{Động lực và nền tảng bài toán}
%%% ============================================================

\begin{frame}{Bối cảnh: từ một tình huống thực tế}
  \footnotesize
  \begin{alertblock}{Câu chuyện mở đầu}
    \footnotesize
    Một nhóm vận hành muốn có một agent AI tự thao tác trên website nội bộ của họ ---
    đặt hàng, cập nhật đơn, trả lời trên diễn đàn. Dùng thẳng một LLM lớn qua API thì
    mỗi lượt gọi đều tốn tiền, phản hồi chậm, và dữ liệu nội bộ phải gửi ra ngoài.
    Chuyển sang một mô hình nhỏ chạy ngay tại máy thì lại vấp bài toán khác: mô hình
    nhỏ chưa biết cách thao tác trên \emph{website cụ thể} của họ, mà để dạy nó thì cần
    dữ liệu huấn luyện --- mà gán nhãn từng bước cho một website thật (AutoWebGLM:
    \textbf{20 người làm liên tục một tháng}) là điều một nhóm nhỏ không kham nổi.
  \end{alertblock}

  \vspace{0.05cm}
  Hai vấn đề khóa luận giải quyết: \textbf{chi phí/độ trễ/quyền riêng tư} khi phụ thuộc
  mô hình lớn, và \textbf{nút thắt dữ liệu} khi tự chủ bằng mô hình nhỏ (SLM ${<}10$B)
  --- trong khi dữ liệu tự thu có thể giảm tới $\approx$\textbf{97\%} chi phí mỗi mẫu
  so với thuê người gán nhãn (Synatra).

  \vspace{0.05cm}
  \begin{block}{Phát biểu bài toán --- và ranh giới của nó}
    \footnotesize
    Tự động khám phá \emph{một} website đang hoạt động và tinh chỉnh SLM để nó hoàn
    thành các yêu cầu bằng ngôn ngữ tự nhiên trên \textbf{chính website đó} --- không
    dùng bất kỳ dữ liệu nào do con người gán nhãn.

    \vspace{0.06cm}
    {\scriptsize $\Rightarrow$ Đích nhắm là \kw{chuyên biệt hóa cho một website cụ thể},
    \emph{không} phải một model tổng quát cho mọi website cùng lĩnh vực. Thứ tái sử
    dụng được là \kw{pipeline}, không phải model (chi tiết ở mục 4).}
  \end{block}
\end{frame}

\begin{frame}{Mục tiêu, câu hỏi nghiên cứu và định vị so với công trình liên quan}
  \begin{columns}[t]
    \begin{column}{0.46\textwidth}
      \scriptsize
      \textbf{Năm mục tiêu nghiên cứu}
      \begin{itemize}\setlength{\itemsep}{1pt}
        \item[\textbf{O1}] Khám phá tự động một website bất kỳ.
        \item[\textbf{O2}] Sinh \textbf{semantic task}, đã qua kiểm chứng.
        \item[\textbf{O3}] Tinh lọc trajectory thành dataset để huấn luyện.
        \item[\textbf{O4}] Fine-tune SLM và đánh giá trên WebArena.
        \item[\textbf{O5}] Đúc kết kinh nghiệm vận hành thực tế.
      \end{itemize}
      \vspace{0.06cm}
      \begin{block}{Bốn câu hỏi nghiên cứu (RQ)}
        \scriptsize
        \begin{itemize}\setlength{\itemsep}{1pt}
          \item[\textbf{CH1}] Pipeline thu được gì, tốn bao nhiêu chi phí?
          \item[\textbf{CH2}] SLM cải thiện như thế nào sau fine-tune?
          \item[\textbf{CH3}] Kết quả so với các agent SOTA ${<}10$B hiện nay?
          \item[\textbf{CH4}] Đóng góp của từng thành phần, điểm nghẽn nằm ở đâu?
